        assert record['path'] == item['path'] and record['sha256'] == item['payload_sha256']
        scratch = retention.safe_path(r, m['scratch']); original = retention.safe_path(r, record['path'])
        assert original.is_relative_to(scratch) and original.suffix in ['.json', '.jsonl']
        assert not original.is_relative_to(r / 'logs') and not retention.secret_name(original)
        if original.exists():
            assert retention.sha_file(original) == record['sha256']
            restored_later.append(record['path'])
        archive = retention.safe_path(r, m['archive'])
        assert archive.is_relative_to(r / 'ops/codex-ops/retention/storage')
        if str(archive) not in archives:
            assert archive.stat().st_size == m['archive_bytes'] and retention.sha_file(archive) == m['archive_sha256']
            archives[str(archive)] = m['archive_sha256']
        assert archives[str(archive)] == m['archive_sha256']
        manifests.append(dict(path=record['path'], batch=item['batch'], stage=phase['stage'], manifest=item['manifest'],
            manifest_sha256=sha(raw), archive=m['archive'], archive_sha256=m['archive_sha256'],
            payload_sha256=record['sha256'], original_bytes=record['bytes']))
