def commit_files(tree,files,label,message,modes=None):
    scan(files,label); parent=git(tree,'rev-parse','HEAD').decode().strip()
    assert not git(tree,'diff','--cached','--name-only').strip()
    index=Path(git(tree,'rev-parse','--path-format=absolute','--git-path','index').decode().strip())
    guard=Path(str(index)+'.lock'); fd=os.open(guard,os.O_CREAT|os.O_EXCL|os.O_WRONLY,0o600)
    try:
        private=A/(label+'-index'); private.write_bytes(index.read_bytes()); env={**os.environ,'GIT_INDEX_FILE':str(private)}
        for path,raw in files.items():
            oid=git(tree,'hash-object','-w','--stdin',data=raw).decode().strip()
            git(tree,'update-index','--add','--cacheinfo',f'{(modes or {}).get(path,"100644")},{oid},{path}',env=env)
        made=git(tree,'write-tree',env=env).decode().strip()
        assert set(git(tree,'diff','--name-only',parent,made).decode().splitlines())==set(files)
        msg=message+'\n\nCo-Authored-By: Codex gpt-6.1-sol <noreply@openai.com>\n'
        commit=git(tree,'commit-tree',made,'-p',parent,data=msg.encode()).decode().strip()
        assert git(tree,'rev-parse','HEAD').decode().strip()==parent
        for path,raw in files.items():
            target=tree/path; target.parent.mkdir(parents=True,exist_ok=True); target.write_bytes(raw)
            target.chmod(0o755 if (modes or {}).get(path)=='100755' else 0o644)
        git(tree,'update-ref','HEAD',commit,parent)
        os.write(fd,private.read_bytes()); os.fsync(fd); os.close(fd); fd=None; os.replace(guard,index)
        return dict(parent=parent,commit=commit,tree=made,paths=list(files))
    finally:
        if fd is not None: os.close(fd)
        if guard.exists(): guard.unlink()
