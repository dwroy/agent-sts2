def allocated(q):
 return sum(f.stat().st_blocks*512 for f in q.rglob('*') if f.is_file() and not f.is_symlink())
def selected(folder):
 files=[]
 for child in folder.iterdir():
  if child.is_file() and not child.is_symlink() and child.suffix=='.index':files.append(child)
  elif child.is_dir() and not child.is_symlink() and child.name in ('record-scan','composition-scan','source-scan'):
   for f in child.rglob('*'):
    if f.is_symlink():raise ValueError('symlink copy retained')
    if f.is_file():files.append(f)
 for f in files:
  ret.safe_path(r,f)
  if ret.secret_name(f):raise ValueError('credential-named path retained without reading')
 return sorted(files)
