def active(scratch):
    text = str(scratch); needle = text.encode()
    for proc in Path('/proc').iterdir():
        if not proc.name.isdigit() or int(proc.name) == os.getpid(): continue
        try:
            cwd = (proc / 'cwd').resolve()
            if cwd == scratch or scratch in cwd.parents: return True
            args = (proc / 'cmdline').read_bytes()
            if needle in args: return True
            if cwd == r or r in cwd.parents or str(r).encode() in args:
                for fd in (proc / 'fd').iterdir():
                    try: target = os.readlink(fd)
                    except OSError: continue
                    if target == text or target.startswith(text + '/'): return True
        except OSError: continue
    return False

def guard(batch, file=None):
    scratch = retention.safe_path(r, batch['scratch'])
    state = retention.load(r / 'ops/codex-ops/learn.json')
    original = state['batches'][batch['batch']]
    assert original.get('state') in ['done', 'failed'] and original.get('rc') == 0
    assert original.get('finished') == batch['finished']
    assert not any(original.get(k) for k in ['learning_request', 'research_request', 'implementation_request', 'feature_request', 'boss_sim', 'checks_pending', 'proposal_repair_needed', 'proposal_audit_errors'])
    for item in state.get('learning_requests', {}).values():
        assert batch['batch'] not in [item.get('original_batch'), item.get('batch'), item.get('verified_batch')]
        assert batch['batch'] not in item.get('attempts', [])
    if batch.get('original_launch'):
        event = retention.safe_path(r, batch['original_launch'])
        with event.open('rb') as stream: first = stream.readline(131073)
        assert sha(first) == batch['original_launch_header_sha256']
    assert not active(scratch), 'active scratch consumer/owner retained'
    if file:
        f = retention.safe_path(r, file['path'])
        assert f.is_relative_to(scratch) and f.is_file() and not f.is_symlink()
        assert f.suffix in ['.json', '.jsonl'] and not retention.secret_name(f)
    return scratch

def payload_verified(m):
    archive = retention.safe_path(r, m['archive']); expected = m['files'][0]
    assert archive.is_relative_to(r / 'ops/codex-ops/retention/storage') and not retention.secret_name(archive)
    key = (str(archive), *identity(archive), m['archive_sha256'])
    if key in verified: return
    assert retention.sha_file(archive) == m['archive_sha256']
    with tarfile.open(archive, 'r:gz') as tar:
        members = tar.getmembers()
        assert len(members) == 1 and members[0].isfile()
        assert members[0].name == expected.get('archive_member_path', expected['path'])
        digest = hashlib.sha256(); total = 0
        with tar.extractfile(members[0]) as stream:
            for chunk in iter(lambda: stream.read(1024**2), b''):
                digest.update(chunk); total += len(chunk)
        assert digest.hexdigest() == expected['sha256'] and total == expected['bytes']
    verified[key] = True

@contextmanager
def serial_gate(batch):
    # Wait for the original retention transaction; do not skip every file during its short lock.
    with (r / 'ops/codex-ops/retention.lock').open('a') as handle:
        fcntl.flock(handle, fcntl.LOCK_EX)
        with retention.locks([r / 'ops/codex-ops' / (batch['batch'] + '.checks.lock')]):
            yield

def manifest_path(file): return storage / 'files' / (sha(file['path'].encode()) + '.manifest.json')
def compress_one(batch, file, restoring=False):
    f = retention.safe_path(r, file['path']); mp = manifest_path(file)
    if not f.exists():
        if mp.exists(): return retention.load(mp)
        raise ValueError('copy already absent; no claimed reclamation')
    with serial_gate(batch):
        scratch = guard(batch, file)
        before = f.stat()
        if not restoring: assert identity(f) == (file['device'], file['inode'], file['bytes'], file['mtime_ns'])
        digest = retention.sha_file(f)
        proof = dict(kind='Roy-authorized-ended48h-process-copy-compression', batch=batch['batch'],
            original_scratch=batch['scratch'], original_launch=batch.get('original_launch'),
            original_launch_header_sha256=batch.get('original_launch_header_sha256'), original_finished=batch['finished'],
            frozen_cutoff_48h=plan['summary']['cutoff_48h'], canonical_state_preserved=batch['original_canonical_state'],
            exp_ingestion=batch.get('exp_ingestion'), stage=file['stage'], plan_sha256=plan_sha,
            root_logs_knowledge_ledgers_reports_proposals_and_original_events_retained=True,
            no_task_completion_ingestion_or_scientific_claim_added=True)
        key = (digest, before.st_size)
        if key not in pool:
            base = storage / 'blobs' / digest
            if base.with_suffix('.manifest.json').exists():
                blob = retention.load(base.with_suffix('.manifest.json'))
                assert blob['files'][0]['sha256'] == digest
            else:
                blob, _ = archive_six(r, base, scratch, [f], proof)
            pool[key] = (blob, str(base.with_suffix('.manifest.json').relative_to(r)))
        blob, original_manifest = pool[key]
        payload_verified(blob)
        member = blob['files'][0].get('archive_member_path', blob['files'][0]['path'])
        item = dict(path=file['path'], archive_member_path=member, bytes=before.st_size,
            sha256=digest, mode=before.st_mode & 0o777, original_mtime_ns=before.st_mtime_ns)
        m = dict(schema='learner-retention-v1', at=retention.stamp(), root=str(r), scratch=batch['scratch'],
            proof=proof, archive=blob['archive'], archive_sha256=blob['archive_sha256'], archive_bytes=blob['archive_bytes'],
            files=[item], state='archived_not_reclaimed', payload_source_manifest=original_manifest,
            identical_payload_deduplicated=member != file['path'], archive_payload_readback_verified=True,
            source_sha_verified=True, original_allocated_bytes=before.st_blocks * 512)
        write(mp, m)
        assert retention.sha_file(f) == digest
        assert identity(f) == (before.st_dev, before.st_ino, before.st_size, before.st_mtime_ns)
        guard(batch, file)
        f.unlink()
        parent = os.open(f.parent, os.O_RDONLY | os.O_DIRECTORY)
        try: os.fsync(parent)
        finally: os.close(parent)
        m.update(state='process_copy_compressed', reclaimed_at=retention.stamp())
        write(mp, m)
        return m
