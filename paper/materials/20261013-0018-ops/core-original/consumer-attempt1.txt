import {readFileSync,writeFileSync,mkdirSync,cpSync,readdirSync,statSync} from 'node:fs';
import {resolve,join,relative} from 'node:path';
import {pathToFileURL} from 'node:url';
import {createHash} from 'node:crypto';
import {execFileSync} from 'node:child_process';
import assert from 'node:assert/strict';
const here=resolve('learner/runs/20261012-core-decklists-learning-v2');
const stage=process.argv[2]??'source';
const repo=resolve(process.argv[3]??'.');
const out=join(here,stage);mkdirSync(out,{recursive:true});
const sha=x=>createHash('sha256').update(x).digest('hex');
const save=(name,x)=>writeFileSync(join(out,name),JSON.stringify(x,null,2)+'\n');
const mod=p=>import(pathToFileURL(join(repo,'agent/src',p+'.ts')).href);
process.env.CHARACTER='silent';
const {setKnowledgeCharacter}=await mod('knowledge/files');setKnowledgeCharacter('silent');
const {KnowledgePrompt}=await mod('brain/knowledge');
const {parseGameState}=await mod('hand/mod/schema');
const {makeKnowledge}=await mod('knowledge/index');
const {RunJournal}=await mod('memory/run-journal');
const changes=JSON.parse(readFileSync(join(here,'changes.json')));
const frames=JSON.parse(readFileSync(join(here,'consumer-frames.json')));
const knowledgeDir=join(out,'knowledge');mkdirSync(knowledgeDir,{recursive:true});
for(const path of ['common','characters/silent'])cpSync(join(repo,'knowledge',path),join(knowledgeDir,path),{recursive:true});
if(stage==='negative')cpSync(join(here,'experience-before.json'),join(knowledgeDir,'characters/silent/experience.json'));
const files=[];
function walk(dir){for(const name of readdirSync(dir)){const path=join(dir,name);if(statSync(path).isDirectory())walk(path);else files.push({path:relative(out,path),sha256:sha(readFileSync(path))});}}
walk(knowledgeDir);
const exp=JSON.parse(readFileSync(join(knowledgeDir,'characters/silent/experience.json')));
// Case-title enrichment is omitted; the production reader and formatter still load the actual experience file.
const prompt=new KnowledgePrompt({postmortems:()=>({path:'fixed-consumer:case-titles-omitted',sections:new Map()})});
const requests=[];
const outputs=[];
for(const frame of frames){
 const raw=frame.state,state=parseGameState(raw);
 const knowledge=makeKnowledge({cards:raw.run.deck.map(c=>({id:c.card_id,type:c.card_type,name:c.name}))},'cache');
 const question={label:frame.label,factsCovered:true,statsCovered:true};
 const memory=new RunJournal().render(state,knowledge,{},question);
 const request={label:frame.label,system:'fixed-consumer input before full knowledge',question:frame.label,payload:{state:raw},memory};
 requests.push({source:frame.source,request});
 const actual=prompt.apply(request,{ascension:10,knowledgeDir});
 const checks=changes.map(c=>({id:c.id,id_found:actual.system.includes('['+c.id+'｜'),addition_found:actual.system.includes(c.added_lesson.trim())}));
 outputs.push({label:frame.label,knowledge:actual.knowledge,checks,request:actual});
}
save('input.json',{character:'silent',ascension:10,repo,code_commit:execFileSync('git',['-C',repo,'rev-parse','HEAD'],{encoding:'utf8'}).trim(),experience_version:exp.version,experience_sha256:sha(readFileSync(join(knowledgeDir,'characters/silent/experience.json'))),knowledge_files:files,requests,limits:'固定真实状态调用KnowledgePrompt.apply全量前缀，不调用模型/执行动作；案例标题装饰省略。'});
save('output.json',{outputs,renders:prompt.renders,natural_consumer:false});
assert(outputs.every(o=>o.knowledge?.mode==='full'),'full knowledge fallback');
assert(outputs.every(o=>o.checks.every(c=>c.id_found&&c.addition_found)),'new bounded lessons missing from consumer');
save('verification.json',{status:'passed',kind:'fixed-consumer',stage,experience_version:exp.version,questions:outputs.length,entries:changes.map(c=>c.id),checks:outputs.flatMap(o=>o.checks),natural_adoption:false});
console.log(JSON.stringify({stage,status:'passed',questions:outputs.length,entries:changes.length,experience_version:exp.version}));
