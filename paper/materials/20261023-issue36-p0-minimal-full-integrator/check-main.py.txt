from pathlib import Path
import hashlib,json,os,subprocess,time
R=Path('/home/dw/Projects/agent-sts2');B=Path(__file__).resolve().parent
A=json.loads((B/'main-assembly.json').read_text());W=Path(A['worktree'])
# Compatibility check can precede full; integration still requires the unique full result.
def git(*a):return subprocess.check_output(['git',*a],cwd=W)
def now():return subprocess.check_output(['date','-Iseconds'],text=True).strip()
assert git('rev-parse','HEAD').decode().strip()==A['base']
assert git('write-tree').decode().strip()==A['tree'] and not git('diff','--name-only').strip()
for row in A['paths']:
 assert hashlib.sha256((W/row['path']).read_bytes()).hexdigest()==row['sha256'],row['path']
def manifest():
 names=git('ls-files','-z','--cached','--others','--exclude-standard','agent','ops','learner','eval','knowledge','tools','.agents').decode().split('\0')
 return [dict(path=n,sha256=hashlib.sha256((W/n).read_bytes()).hexdigest()) for n in sorted(set(names)) if n and (W/n).is_file() and not (W/n).is_symlink() and not (W/n).name.startswith('.env') and n!='ops/instance-binding.json']
before=manifest();(B/'main-inputs-before.json').write_text(json.dumps(before,indent=2)+'\n')
s=dict(at=now(),state='running',base=A['base'],tree=A['tree'],worktree=str(W),steps=[],full_repeated=False)
def put():(B/'main-checks.json').write_text(json.dumps(s,ensure_ascii=False,indent=2)+'\n')
env={**os.environ,'PATH':str(Path.home()/'.local/node/bin')+':'+os.environ['PATH'],'PYTHONDONTWRITEBYTECODE':'1','OMP_NUM_THREADS':'1','OPENBLAS_NUM_THREADS':'1','MKL_NUM_THREADS':'1','STS2_MINIMAL_P0_COMPATIBILITY':'1'};env.pop('CHARACTER',None)
files=['learner-native-prelude','learner-tui-reconciliation','learner-herdr','learner-herdr-ownership','learner-herdr-launcher','learner-herdr-receipts','boss-sim-home','learner-entry','learner-startup-recovery','brain-view','learner-stdio-view']
commands=[('tsc',['node','node_modules/typescript/bin/tsc','-p','tsconfig.json','--noEmit']),('directed',['node','node_modules/vitest/vitest.mjs','run',*['tests/'+n+'.test.ts' for n in files],'--pool=forks','--maxWorkers=1','--no-file-parallelism','--reporter=default','--reporter=json','--outputFile.json='+str(B/'main-vitest.json')]),('python-tui',['python3','-B','-m','unittest','discover','-s','../ops/tests','-p','test_learner_tui.py']),('syntax',['bash','-n','../ops/codex-ops-learner.sh'])]
put()
try:
 for name,args in commands:
  step=dict(name=name,argv=['nice','-n','19',*args],started_at=now());s['steps'].append(step);start=time.monotonic();put()
  with (B/('main-'+name+'.log')).open('xb') as out:
   p=subprocess.run(step['argv'],cwd=W/'agent',env=env,stdout=out,stderr=subprocess.STDOUT)
  step.update(exit_code=p.returncode,seconds=round(time.monotonic()-start,3));put();assert p.returncode==0,name+' failed'
 after=manifest();(B/'main-inputs-after.json').write_text(json.dumps(after,indent=2)+'\n');assert before==after
 assert git('write-tree').decode().strip()==A['tree'] and not git('diff','--name-only').strip()
 q=json.loads((B/'main-vitest.json').read_text());assert q['success']
 s.update(state='passed',inputs_unchanged=True,input_count=len(before),vitest=dict(files=len(q['testResults']),passed=q['numPassedTests'],failed=q['numFailedTests'],pending=q['numPendingTests']))
except BaseException as e:s.update(state='failed',reason=str(e));raise
finally:s['ended_at']=now();put()
