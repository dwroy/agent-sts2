sha = lambda raw: hashlib.sha256(raw).hexdigest()
load = lambda path: json.loads(path.read_bytes())
def write(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')
def git(*args, cwd=r, env=None, data=None):
    return subprocess.check_output(['git', '-C', str(cwd), *args], input=data, env=env, stderr=subprocess.PIPE)
def blob(ref, path):
    return git('show', ref + ':' + path)

lock_path = r / 'ops/live-merge.lock'
with lock_path.open('a') as lock:
    try:
        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        acquired = True
    except BlockingIOError:
        acquired = False
    inode = str(lock_path.stat().st_ino)
    lock_lines = [line for line in Path('/proc/locks').read_text().splitlines()
                  if len(line.split()) > 5 and line.split()[5].split(':')[-1] == inode]
    holders = []
    for line in lock_lines:
        pid = int(line.split()[4])
        try:
            proc = Path('/proc') / str(pid)
            args = (proc / 'cmdline').read_bytes().split(b'\0')
            assert b'recheck' in args and bid.encode() in args
            holders.append(dict(pid=pid, birth_ticks=(proc / 'stat').read_text().split(') ', 1)[1].split()[19],
                                command=[x.decode() for x in args if x], original_batch=True))
        except FileNotFoundError:
            holders.append(dict(pid=pid, already_exited=True))
    receipt = dict(at=at, nonblocking=True, acquired=acquired, exit=0 if acquired else 75,
        actual_delivery_registered_before_gate=True, lock=str(lock_path.relative_to(r)),
        original_lock_lines=lock_lines, holders=holders, main_live_unmodified=True,
        tests_or_new_model_not_started=True, existing_original_host_result=None,
        scope='One gate attempt only; do not wait, poll or steal the existing original exact-tree check lock.')
    write(pub / 'live-lock-gate.json', receipt)
assert not acquired, 'Gate is clear: continue original authorized integration rather than ending pending.'
