WANTED = set(json.loads((SCRATCH / 'dispatch.json').read_text())['runs'])


def save(name, value):
    (SCRATCH / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')


def run_id(row):
    return row.get('run_id') or row.get('state', {}).get('run_id') or row.get('run')


manifest = []
datasets = {}
for entry in json.loads((PRIOR / 'raw-input-manifest.json').read_text()):
    source = Path(entry['source'])
    target = SCRATCH / Path(entry['frozen']).name
    digest = hashlib.sha256()
    counts = collections.Counter()
    rows = {}
    index = []
    with source.open('rb') as src, target.open('wb') as dst:
        for item in entry['index']:
            src.seek(item['offset'])
            raw = src.readline()
            assert hashlib.sha256(raw).hexdigest() == item['sha256'], (source, item['line'])
            row = json.loads(raw)
            run = run_id(row)
            assert run == item['run'] and run in WANTED, (source, item['line'], run)
            if source.name == 'states.jsonl':
                identity = row['state']['run']
                assert identity['character_id'] == 'SILENT' and identity['ascension'] == 10
            dst.write(raw)
            digest.update(raw)
            counts[run] += 1
            index.append({**item, 'bytes': len(raw)})
            rows[item['line']] = row
    assert digest.hexdigest() == entry['sha256'] and dict(counts) == entry['counts']
    manifest.append({**entry, 'frozen': str(target), 'index': index, 'verified_rows': len(index)})
    datasets[source.stem] = rows
    print(source.name, len(rows), dict(counts), '逐行原字节、身份和SHA通过')
save('raw-input-manifest.json', manifest)

def enemy_hp(turn):
    return sum(int(hp) for hp in re.findall(r' (\d+)/\d+(?:,|$)', turn['enemies']))


def actual(run, floor, attempt, turn):
    matches = [(n, row) for n, row in attempts.items() if row['run_id'] == run
               and row['floor'] == floor and row['attempt'] == attempt]
    assert len(matches) == 1, (run, floor, attempt)
    line, row = matches[0]
    turns = {t['turn']: t for t in row['summary']['turns']}
    current, following = turns[turn], turns[turn + 1]
    return {'sl_line': line, 'attempt': attempt, 'turn': turn, 'hp_before': current['hp'],
            'hp_next_turn': following['hp'], 'hp_loss': current['hp'] - following['hp'],
            'enemy_before': enemy_hp(current), 'enemy_next_turn': enemy_hp(following),
            'enemy_net_loss': enemy_hp(current) - enemy_hp(following),
            'plays': current['plays'], 'later_plays': following['plays']}

def snapshot(line):
    row = states[line]
    s = row['state']
    player = s.get('player') or {}
    combat = s.get('combat') or {}
    return {'line': line, 'run': s['run_id'], 'floor': s['run']['floor'], 'turn': s.get('turn'),
            'hp': player.get('current_hp'), 'block': player.get('block'),
            'powers': player.get('powers'), 'slots': player.get('potions'),
            'selection': s.get('selection'), 'pending_card': combat.get('pending_card'),
            'enemies': s.get('enemies') or combat.get('enemies')}
