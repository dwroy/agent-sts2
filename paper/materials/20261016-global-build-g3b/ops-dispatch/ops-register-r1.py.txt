def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def date():
    return subprocess.check_output(['date', '+%Y-%m-%d %H:%M:%S %Z'], text=True).strip()


def save(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    temp = path.with_name(path.name + '.ops-' + str(os.getpid()) + '.tmp')
    temp.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')
    os.replace(temp, path)


def ref(path):
    path = Path(path)
    raw = path.read_bytes()
    return dict(path=str(path.relative_to(root)), bytes=len(raw), sha256=sha(raw))


def alive(pid):
    if not isinstance(pid, int) or pid <= 0:
        return False
    try:
        os.kill(pid, 0)
        return True
    except ProcessLookupError:
        return False


def verify():
    assert ref(phase / 'learner-input.md')['sha256'] == 'aff9f3238929dd5ef41accfaafc6ebc778b209bbeeb05d63493938f53be49e9f'
    manifest_ref = ref(phase / 'inputs/manifest.json')
    assert manifest_ref['sha256'] == '588276ef299dd6802bfe3e7e69e5ad07a506a91eb81bd034bdf30b10824d6ecb'
    manifest = json.loads((phase / 'inputs/manifest.json').read_bytes())
    coordination = ref(phase / 'coordination/anchor-index.json')
    assert coordination['sha256'] == '724e2996f7634079901a6ff25838f4a15b5e8a2b85a012714ae500015a40791d'
    artifacts = []
    for name, expected in manifest['artifacts'].items():
        item = ref(phase / 'inputs' / name)
        assert all(item[key] == expected[key] for key in ['bytes', 'sha256'])
        artifacts.append(item)
    count = 0
    with (root / 'logs/decisions.jsonl').open('rb') as source:
        for item in manifest['raw_construction_record_index']:
            assert item['path'] == 'logs/decisions.jsonl'
            assert item['off'] + item['len'] <= manifest['source_bounds']['decisions']['complete_prefix_bytes']
            source.seek(item['off'])
            raw = source.read(item['len'])
            assert sha(raw) == item['sha256']
            count += 1
    assert count == 1371
    bounds = []
    for name, item in manifest['source_bounds'].items():
        st = (root / item['path']).stat()
        assert st.st_size >= item['complete_prefix_bytes']
        assert st.st_ino == item['inode'] and st.st_dev == item['device']
        bounds.append(dict(stream=name, **item, current_size_at_ops_read=st.st_size))
    original = tree / 'learner/runs/20261012-024330-strategy-proposal.jsonl'
    original_ref = ref(original)
    assert original_ref['sha256'] == '05f714b76491fa0faf2872b2a2011b1efe8d8c4c8e8e66244949b31d7ff25687'
    launch = []
    threads = set()
    for raw in original.read_bytes().splitlines():
        event = json.loads(raw)
        if event.get('type') == 'learner_launch':
            launch.append(event)
        if event.get('type') == 'thread.started':
            threads.add(event.get('thread_id'))
    assert len(launch) == 1 and launch[0]['engine'] == 'codex' and launch[0]['cwd'] == str(tree)
    assert threads == {thread}
    command = launch[0]['command']
    original_model = command[command.index('--model') + 1]

    notice = json.loads((state_dir / 'learning-notices/d70532ef14bf73d453bdbc83.json').read_text())
    stages = {}
    for key in ['ops_forwarded', 'watcher_received', 'roy_notified']:
        item = notice[key]
        path = Path(item['path'])
        if not path.is_absolute():
            path = root / path
        assert path.resolve().is_relative_to(root) and sha(path.read_bytes()) == item['sha256']
        stages[key] = item
    result = dict(at=date(), manifest=manifest_ref, input=ref(phase / 'learner-input.md'),
        artifacts=artifacts, indexed_original_rows_SHA_verified=count, source_bounds=bounds,
        optional_coordination_index=coordination, coordination_is_metadata_only_not_labels=True,
        game_knowledge_labels_derived_by_ops=False, original_owner_log=original_ref,
        original_thread=thread, original_model_history_retained=original_model,
        requested_resume_model='gpt-6.1-sol', effort='xhigh', service_tier='default',
        existing_learner_wrapper_and_permissions_retained=True, d705_existing_stages=stages,
        no_duplicate_notice_or_role_ACK=True, Roy_ACK=None)
