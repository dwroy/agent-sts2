import hashlib

out = Path(__file__).parent
manifest = json.loads((out / 'evidence-manifest.json').read_text())
rows = {}
for name, meta in manifest.items():
    raw = (out / (name + '.jsonl')).read_bytes()
    assert hashlib.sha256(raw).hexdigest() == meta['selected_sha256']
    lines = raw.splitlines(keepends=True)
    assert len(lines) == len(meta['entries'])
    rows[name] = {}
    for entry, line in zip(meta['entries'], lines):
        assert hashlib.sha256(line).hexdigest() == entry['sha256']
        value = json.loads(line)
        run = value.get('run_id') or value.get('run') or value.get('state', {}).get('run_id')
        assert run == entry['run']
        if name == 'states':
            assert value['state']['run']['character_id'].lower() == 'silent'
            assert value['state']['run']['ascension'] == 10
        rows[name][entry['line']] = value

attempts = {(v['run_id'], v['floor'], v['attempt']): (n, v)
            for n, v in rows['sl-attempts'].items()}

def enemy_hp(text):
    match = re.search(r' (\d+)/(\d+)', text)
    assert match is not None
    return int(match.group(1))

def pair(run, floor, turn, attempt_a, attempt_b):
    result = {'run': run, 'floor': floor, 'turn': turn, 'sides': []}
    for attempt in (attempt_a, attempt_b):
        source_line, row = attempts[(run, floor, attempt)]
        summary = {t['turn']: t for t in row['summary']['turns']}
        before, after = summary[turn], summary[turn + 1]
        candidates = [(n, v) for n, v in rows['decisions'].items()
                      if v['run_id'] == run and v['floor'] == floor and v['turn'] == turn
                      and v.get('sl_attempt') == attempt and v.get('decider') == 'jev'
                      and 'plan' in v.get('questions', {})]
        assert candidates
        dn, decision = candidates[0]
        result['sides'].append({'attempt': attempt, 'sl_line': source_line,
            'decision_line': dn, 'fingerprint': decision['fingerprint'],
            'hp_before': before['hp'], 'hp_after': after['hp'],
            'net_hp_loss': before['hp'] - after['hp'],
            'net_enemy_hp_reduction': enemy_hp(before['enemies']) - enemy_hp(after['enemies']),
            'plays': before['plays'], 'next_turn_plays': after['plays'],
            'sl_choice': decision.get('sl_explore'),
            'full_fight_result': row['result'], 'last_turn': row['turns']})
    a, b = result['sides']
    result['same_logged_fingerprint'] = a['fingerprint'] == b['fingerprint']
    assert result['same_logged_fingerprint']
    result['delta'] = {k: b[k] - a[k] for k in ('net_hp_loss', 'net_enemy_hp_reduction')}
    result['controlled_whole_fight'] = False
    return result

def state_summary(n):
    s = rows['states'][n]['state']
    c = s.get('combat') or {}
    p = c.get('player') or {}
    return {'line': n, 'run': s['run_id'], 'floor': s['run']['floor'], 'turn': s['turn'],
        'hp': p.get('current_hp', s['run']['current_hp']), 'block': p.get('block'),
        'potions': s['run']['potions'], 'powers': p.get('powers'),
        'enemies': c.get('enemies'), 'hand': c.get('hand')}
