from pathlib import Path
import fcntl, hashlib, json, os, re, subprocess, sys

r = Path.cwd(); live = r / '.worktrees/live'
p = r / 'ops/codex-ops/record-staging/20261017-1252'
pub = r / 'paper/materials/20261017-1252-ops'
sha = lambda raw: hashlib.sha256(raw).hexdigest()
load = lambda f: json.loads(f.read_bytes())
def write(f, v): f.write_text(json.dumps(v, ensure_ascii=False, indent=2) + '\n')
def git(*a, env=None, data=None):
    return subprocess.check_output(['git', *a], env=env, input=data, stderr=subprocess.PIPE)
def blob(ref, path):
    answer = subprocess.run(['git', 'show', ref + ':' + path], capture_output=True)
    assert answer.returncode in [0, 128]
    return answer.stdout if answer.returncode == 0 else None
def stage(path, raw, env):
    oid = git('hash-object', '-w', '--stdin', data=raw).decode().strip()
    git('update-index', '--add', '--cacheinfo', '100644', oid, path, env=env)
def physical(path): return (r / path).read_bytes() if (r / path).is_file() else None

sys.path.insert(0, str(r / 'ops'))
import learning_requests as lr

assert not (pub / 'main-source-sync.json').exists()
host = load(pub / 'combined-host-terminal.json')
publication = load(r / 'paper/materials/20261017-1205-ops/h0-original-publication.json')
source = publication['source']; base = git('rev-parse', source + '^').decode().strip()
assert base == '54e36d0830ab640130557b62a078b25954912ff4'
core = ['agent/src/reflex/rollout.ts', 'agent/src/reflex/rollout-live.ts',
        'agent/tests/silent-boss-weak.test.ts']
experience_path = 'knowledge/characters/silent/experience.json'

    'paper/materials/20261017-human-pace-h0/', 'experiments/boss-sim/silent/'))]
assert set(publication['owned_paths']) == set(core + artifacts + [experience_path, 'paper/materials/decision-log.md'])
with (r / 'ops/live-merge.lock').open('a') as lock:
    try:
        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
    except BlockingIOError:
        at = subprocess.check_output(['date', '+%Y-%m-%d %H:%M:%S %Z'], text=True).strip()

    head = git('rev-parse', 'HEAD').decode().strip()

    assert [x for x in main_versions['versions'] if x.get('release_id') in ['V5-R381', 'V5-R382']] == [x for x in live_versions['versions'] if x.get('release_id') in ['V5-R381', 'V5-R382']]
    assert not git('ls-files', '-u').strip()
    original_index_entries = git('ls-files', '--stage', '-z')
    idx = Path(git('rev-parse', '--git-path', 'index').decode().strip())
    idx = idx if idx.is_absolute() else r / idx

    before = {}

    assert applied.returncode == 0 and not git('ls-files', '-u', env=private).strip()
    temp_tree = git('write-tree', env=private).decode().strip()
    values = {}; leaves = []
    norm = lambda raw: re.sub(rb'\b(?:Roy|Dai)\b', b'Roy', raw)

        leaves.append(dict(path=path, old_main_sha256=sha(before[path]) if before[path] is not None else None,
