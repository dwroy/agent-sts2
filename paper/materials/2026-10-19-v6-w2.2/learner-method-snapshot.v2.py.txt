def obs(f):
    x=dict(run_id=f['run_id'],floor=f['floor'])
    if isinstance(f.get('turn'),int):x['turn']=f['turn']
    return x

def stop(rid):
    candidates=[]
    p=INV[rid].get('first_predicted_death')
    if p:candidates.append(dict(ts=p['ts'],floor=p['floor'],turn=p.get('turn'),kind='predicted_death',source=p['ref']))
    for x in SL[rid]:
        if x['row']['result']=='died':candidates.append(dict(ts=x['row'].get('ended_at') or x['row']['ts'],floor=x['row']['floor'],turn=x['row'].get('turns'),kind='died',source=x['ref']))
    return min(candidates,key=lambda x:x['ts']) if candidates else None

STOPS={rid:stop(rid) for rid in RUNS}

def within(rid,ts,attempt):
    return attempt=='including_sl' or STOPS[rid] is None or ts<=STOPS[rid]['ts']

def stratum(rid):

    version=r['code']+';records='+COV[rid]['engine_observed']+';accepted='+i['actual_engine_group']+';config='+','.join(i['config_commits'])
    assert len(version)<=500
    return r['ascension'],version

def percentile(v,q):
    v=sorted(v);pos=(len(v)-1)*q;a=int(pos);z=min(a+1,len(v)-1)
    return v[a]+(v[z]-v[a])*(pos-a)

def statistic(values,method,unit='independent_run',aggregation='median',unknown='无满足本指标观察条件的独立局；不补零',distribution=False):
    values=[(rid,float(value),e) for rid,value,e in values if value is not None]
    n=len(values)
    if not n:return dict(n=0,unit=unit,method=method,value=None,se=None,distribution=[],evidence=[],unknown_reason=unknown,small_sample=True)
    assert len({rid for rid,_,_ in values})==n
    nums=[v for _,v,_ in values]
    if aggregation=='median':value=statistics.median(nums)
    elif aggregation=='mean':value=statistics.mean(nums)
    elif aggregation=='sum':value=sum(nums)
    elif aggregation=='min':value=min(nums)
    elif aggregation=='max':value=max(nums)
    else:raise ValueError(aggregation)
    se=None
    if aggregation=='mean' and n>=2:se=statistics.stdev(nums)/math.sqrt(n)
    elif aggregation=='median' and n>=3:
        rng=random.Random(20261019)
        se=statistics.stdev(statistics.median(rng.choices(nums,k=n)) for _ in range(400))
    ev=list({json.dumps(e,sort_keys=True):e for _,_,e in values}.values())
    dist=[dict(label=str(k),count=v) for k,v in sorted(collections.Counter(nums).items())] if distribution else []
    r=dict(n=n,unit=unit,method=method+('; SE=400次固定种子20261019独立局bootstrap' if aggregation=='median' and n>=3 else '; n<3中位数SE未知' if aggregation=='median' else '; 计数/极值不报告抽样SE' if aggregation in ['sum','min','max'] else '; 均值SE按独立局'),value=value,se=se,distribution=dist,evidence=ev,unknown_reason=None,small_sample=n<10)
    if aggregation in ['median','mean','min','max']:r.update(p25=percentile(nums,.25),p75=percentile(nums,.75))
    return r

def rate(values,method,unknown='无可用独立局分母，胜率未知'):
    r=statistic(values,method+'；观察性比例，边界SE=0不表示确定，Wilson区间见metric-samples',aggregation='mean',unknown=unknown)
    if r['n']:
        p=r['value'];n=r['n'];r['se']=math.sqrt(p*(1-p)/n)
    return r

def initial_obs(rid):
    return obs(HISTORY[rid][0])

def at_time(rid,ts):
    fs=HISTORY[rid];ix=bisect.bisect_right([x['ts'] for x in fs],ts)-1
    return fs[ix] if ix>=0 else None

def classify(b):
    if b['sl']:
        k=b['sl']['row'].get('fight_kind');return 'normal' if k=='hallway' else k
    kinds={x['node']['node_type'] for x in MAPS[(b['run_id'],b['floor'])]}
    return {'Monster':'normal','Elite':'elite','Boss':'boss'}.get(next(iter(kinds))) if len(kinds)==1 else 'unknown'

def turns_after(b,a):
    result=[]
    turns=b['turns']
    for i,t in enumerate(turns[:-1]):
        u=turns[i+1];f=t.get('ready_start');g=u.get('ready_start')
        if not f or not g or t['turn']<=a['turn'] or u['turn']!=t['turn']+1:continue
        dep=hp_decrease(f,g)
        if dep is None or t['phase_or_heal']:continue
        result.append(dict(turn=t['turn'],start_ts=f['ts'],end_ts=g['ts'],hp_damage=dep,block_lower_bound=t['block_positive'],hp_loss=max(0,f['hp']-g['hp']) if not t.get('hp_positive_heal') and f['hp']>=g['hp'] else None,enemies=len(f['enemies']),start_source=f['source'],end_source=g['source'],evidence=obs(f)))
    return result

def clean_event(e):
    return dict(run_id=e['run_id'],floor=e['floor'],turn=e['turn'],ts=e['ts'],card=e['card'],delta_block=e['delta_block'],hp_depletion=e['hp_depletion'],energy_net=e['energy_net'],before_source=e['before']['source'],after_source=e['after']['source'])
