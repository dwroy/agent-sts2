## Rebuild (after each run)

```
export PATH=$HOME/.local/node/bin:$PATH
# 1. model + gates + calibration (src/knowledge/fight-value.json, fight-value-gates.json, notes/fight-value-backtest.md), ~5 min
python3 tools/build-fight-value.py all      # extract logs -> .cache/fight-value-rows.jsonl, then train
# 2. optional backtest of the rollout (this file), ~5 min
python3 tools/build-fight-value.py folds --fold-dir /tmp/rb/folds
npx tsx tools/rollout-backtest.ts extract --work /tmp/rb
for i in $(seq 0 15); do npx tsx tools/rollout-backtest.ts run --work /tmp/rb --shard $i --shards 16 --fold-dir /tmp/rb/folds & done; wait
mkdir -p /tmp/rb/solo && ln -sf /tmp/rb/decisions.jsonl /tmp/rb/solo/ && npx tsx tools/rollout-backtest.ts run --work /tmp/rb/solo --shard 3 --shards 8 --fold-dir /tmp/rb/folds
python3 tools/build-fight-value.py rollout-report --work /tmp/rb
```
