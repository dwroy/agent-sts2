from pathlib import Path
import fcntl
import hashlib
import json
import os
import subprocess

r = Path('/home/dw/Projects/agent-sts2')
live = r / '.worktrees/live'
p = r / 'ops/codex-ops/record-staging/20261019-0811'
pub = r / 'paper/materials/20261019-0811-ops'
candidate = 'd5f0704ffae5f94254c6ef79124bde808ec63fcf'
base_live = 'b69c5f58d8651b164e54bcb4e6dbcd2e708ac058'
prepared = json.loads((r / 'paper/materials/20261018-2212-ops/prepared-main-sync-groups.json').read_text())

def git(*args, cwd=r, env=None, data=None):
    return subprocess.check_output(['git', '-C', str(cwd), *args], env=env, input=data, stderr=subprocess.PIPE)

def blob(ref, name):
    return git('show', ref + ':' + name)

def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def save(path, value):
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')

def stage(index_env, name, raw, mode='100644'):
    oid = git('hash-object', '-w', '--stdin', data=raw).decode().strip()
    git('update-index', '--add', '--cacheinfo', mode, oid, name, env=index_env)

def entry(ref, name):
    raw = git('ls-tree', ref, '--', name).decode().strip()
    return raw.split()[0] if raw else None

def commit(tree, parents, title):
    args = ['commit-tree', tree]
    for parent in dict.fromkeys(parents):
        args.extend(['-p', parent])
    return git(*args, data=(title + '\n\nCo-Authored-By: Codex gpt-6.1-sol <noreply@openai.com>\n').encode()).decode().strip()

def install(cwd, name, raw, mode):
    dest = cwd / name
    dest.parent.mkdir(parents=True, exist_ok=True)
    assert not dest.is_symlink()
    tmp = dest.with_name(dest.name + '.storage-publish.tmp')
    assert not tmp.exists()
    tmp.write_bytes(raw)
    os.chmod(tmp, 0o755 if mode == '100755' else 0o644)
    os.replace(tmp, dest)
