import hashlib
import json
from pathlib import Path

OUT = Path(__file__).resolve().parent
checks = []


def check(label, actual, expected):
    assert actual == expected, (label, actual, expected)
    checks.append({'name': label, 'actual': actual, 'expected': expected})


for name in ['decisions', 'states']:
    index = json.loads((OUT / (name + '.jsonl.index.json')).read_text())
    with Path(index[0]['source']).open('rb') as source:
        for item in index:
            source.seek(item['offset'])
            raw = source.read(item['bytes'])
            assert hashlib.sha256(raw).hexdigest() == item['sha256'], item
            row = json.loads(raw)
            assert (row.get('run_id') or row.get('state', {}).get('run_id')) == item['run_id']
    checks.append({'name': name + '原日志逐offset/length/SHA读回', 'rows': len(index), 'verified': True})

assigned = json.loads((OUT / 'assigned-proposals.json').read_text())
for n, proposal in enumerate(assigned, 1):
    check(proposal['add']['id'] + '原提案SHA',
          hashlib.sha256((OUT / f'original-{n:02}.md').read_bytes()).hexdigest(),
          proposal['add']['proposal_sha256'])

facts = {row['line']: row for group in json.loads((OUT / 'state-facts.json').read_text()).values() for row in group}


def player(line, key, default=None):
    return facts[line]['player_powers'].get(key, default)


def enemy(line, key, index=0):
    return facts[line]['enemies'][index][key]


def power(line, key, index=0, default=None):
    return enemy(line, 'powers', index).get(key, default)


def attack(line, index=0):
    return sum(v.get('total_damage') or 0 for v in enemy(line, 'intents', index) or [])


check('DUZ F27 T4污染技能/能力/技能', [player(n, 'TAINTED_POWER', 0) for n in [274830,274831,274832,274833]], [0,3,3,6])
check('DUZ F27 T5火花增长与污染', [power(274834,'VITAL_SPARK_POWER'),*[player(n,'TAINTED_POWER',0) for n in [274834,274836,274837,274838]]], [6,0,6,12,18])
check('5PM F38电击成长', [power(274308,'STRENGTH_POWER',4),attack(274308,4),power(274313,'STRENGTH_POWER',3),attack(274313,3)], [2,17,4,19])
check('5PM电击类型同但临时索引重排', [enemy(274308,'id',4),enemy(274313,'id',3),enemy(274308,'max_hp',4),enemy(274313,'max_hp',3)], ['ZAPBOT','ZAPBOT',23,23])
check('5PM原始敌人无持久实例字段', any(e['identity_fields'] for f in facts.values() for e in f['enemies']), False)
check('5PM F39失落退场力量端点', [player(274351,'STRENGTH_POWER',0),player(274358,'STRENGTH_POWER',0)], [-4,0])
check('CA F13加压力量/攻击', [power(n,'STRENGTH_POWER',0,0) for n in [273237,273252,273263]], [0,4,8])
check('CA F13喷水/弱', [attack(n) for n in [273237,273252,273254,273263]], [11,15,11,14])
check('CA F13覆甲现场层数', [power(n,'PLATING_POWER') for n in [273237,273247,273252,273258,273263]], [9,9,8,7,6])
check('YF F48阶段端点与旧毒', [enemy(277512,'hp'),power(277512,'POISON_POWER'),enemy(277513,'hp'),power(277513,'POISON_POWER',0,0)], [4,4,212,0])
check('YF已建触媒但当前零毒', [player(277519,'ACCELERANT_POWER'),power(277519,'POISON_POWER',0,0)], [2,0])
check('YF苦无敏捷变化与已有挡', [player(277525,'DEXTERITY_POWER'),player(277526,'DEXTERITY_POWER'),facts[277525]['block'],facts[277526]['block']], [5,6,23,23])
check('XP普通镣铐临时减力/恢复', [power(278236,'STRENGTH_POWER',0,0),power(278237,'STRENGTH_POWER'),attack(278237),power(278242,'STRENGTH_POWER',0,0)], [0,-9,0,0])
check('XP船夹板轮初与牌后挡', [facts[278236]['block'],facts[278240]['block']], [14,20])
check('XP朝向与虚弱分列', [attack(n,1) for n in [278249,278252,278253]], [57,38,28])
for run, pairs in [('8JRE1C4H4Z2W', [(1,2,-3,-10),(3,2,-15,-17)]),
                   ('XP2SL33HT0D9', [(1,3,-9,-24),(2,3,-14,-27)])]:
    audit = json.loads((OUT / (run + '-scoped-resolved-audit.json')).read_text())
    for attempt, turn, hp, damage in pairs:
        found = next(row for row in audit['observed_turns'] if row['floor']==33 and row['attempt']==attempt and row['turn']==turn)
        check(f'{run} F33 attempt{attempt} T{turn}观测端点',
              [found['facts']['player_hp_delta'],sum(e['hp_delta'] for e in found['facts']['enemy_hp_changes'])], [hp,damage])

receipts = json.loads((OUT / 'consumer-receipts-resolved.json').read_text())
check('六个限定输入消费者实际rc', [row['rc'] for row in receipts if row['scope']=='scoped'], [0]*6)
check('两局缺decision_id完整输入保留拒绝', [row['rc'] for row in receipts if row['scope']=='full'], [1,1])
check('完整候选执行没有被冒证', all(row['all_full_candidate_results_unknown'] for row in receipts if row['scope']=='scoped'), True)
(OUT / 'evidence-verification.json').write_text(json.dumps({'kind':'evidence_integrity_and_existing_consumer_checks_not_tsc_vitest', 'checks':checks},ensure_ascii=False,indent=2)+'\n')
print('证据核验通过', len(checks), '项；不是源码红绿或tsc/vitest测试。')
