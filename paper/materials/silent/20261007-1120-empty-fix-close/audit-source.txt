import importlib.util
import json
import subprocess
from pathlib import Path

root = Path('/home/dw/Projects/agent-sts2')
worktree = root / '.worktrees/codex-dev'
out = root / 'learner/runs/20261007-111302-fix-batch'
prior = worktree / 'paper/materials/silent/20261007-1051-boss-exp69-release/empty-fix/already-fixed.json'

def git(*args):
    return subprocess.run(['git', '-C', str(worktree), *args], text=True, capture_output=True, check=False)

base = git('rev-parse', 'HEAD').stdout.strip()
live = git('rev-parse', 'live').stdout.strip()
rows = json.loads(prior.read_text())
for row in rows:
    row['ancestry'] = {ref: git('merge-base', '--is-ancestor', row['commit'], ref).returncode == 0
                       for ref in (base, 'main', live)}
    assert all(row['ancestry'].values()), row
queue = (root / 'notes/fix-queue-v4.md').read_text()
active = [i for i, line in enumerate(queue.splitlines(), 1) if line.startswith('- [ ]')]
source_paths = ['agent/src', 'agent/tests', 'agent/tools', 'agent/package.json', 'agent/tsconfig.json',
                'learner/*.ts', 'learner/*.py', 'eval/*.py', 'eval/*.ts', 'eval/cost-config.json', 'knowledge/builders']
source_diff = git('diff', '--name-only', base, live, '--', *source_paths).stdout.splitlines()
assert not source_diff, source_diff
spec = importlib.util.spec_from_file_location('learner_checks', worktree / 'ops/learner_checks.py')
module = importlib.util.module_from_spec(spec)
spec.loader.exec_module(module)
completion_diff = git('diff', '--name-only', base, live, '--', *module.SOURCE_PATHS).stdout.splitlines()
status = git('status', '--porcelain', '--untracked-files=all').stdout
assert not status, status
proof = {'base': base, 'head': base, 'live': live, 'already_fixed_count': len(rows), 'unchecked_queue_lines': active,
         'source_diff': source_diff, 'completion_source_diff': completion_diff, 'worktree_clean': not status,
         'fixes': [], 'merged': None,
         'note': 'No new fix commits. Independent scheduler/task changes differ from live; do not fabricate a merge or overwrite parallel work.'}
(out / 'already-fixed.json').write_text(json.dumps(rows, ensure_ascii=False, indent=2) + '\n')
md = ['# 既有修复核验', '', f'基线：{base}', f'只读核验 live：{live}', '', '| 条目 | 修复提交 | 本分支／main／live |', '|---|---|---|']
md.extend(f"| {row['item']} | {row['commit']} | 已包含 |" for row in rows)
(out / 'already-fixed.md').write_text('\n'.join(md) + '\n')
(out / 'audit.json').write_text(json.dumps(proof, ensure_ascii=False, indent=2) + '\n')
print(json.dumps(proof, ensure_ascii=False, indent=2))
