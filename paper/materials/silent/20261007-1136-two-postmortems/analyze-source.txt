import json,pathlib,collections,datetime,bisect
P=pathlib.Path('learner/runs/20261007-111301-postmortem')
def compact(s):
 c=s.get('combat') or {};p=c.get('player') or {};run=s.get('run') or {}
 return {'floor':run.get('floor'),'turn':s.get('turn'),'hp':p.get('current_hp',run.get('current_hp')),'block':p.get('block'),'energy':p.get('energy'),'powers':[(x['power_id'],x['amount']) for x in p.get('powers',[])], 'enemies':[(e['index'],e['enemy_id'],e['current_hp'],e.get('block'),e['is_alive'],e.get('move_id'),[(v['power_id'],v['amount']) for v in e.get('powers',[])],sum(i.get('total_damage') or 0 for i in e.get('intents',[]))) for e in c.get('enemies',[])]}
for rid in ['2Y27VAYZDA02','TDLBRNA0R05B']:
 ds=[json.loads(x) for x in (P/(rid+'-decisions.jsonl')).open()];ss=[json.loads(x) for x in (P/(rid+'-states.jsonl')).open()]
 byobs={x['observed_ts']:x['state'] for x in ss}; times=[x['observed_ts'] for x in ss]
 def state(d):
  if d['observed_ts'] in byobs:return byobs[d['observed_ts']]
  ix=bisect.bisect_right(times,d['observed_ts'])-1
  return ss[max(0,ix)]['state']
 with (P/(rid+'-summary.txt')).open('w') as o:
  def out(*a):print(*a,file=o)
  out('局号',rid,'决策',len(ds),'状态',len(ss),'用时秒',(datetime.datetime.fromisoformat(ds[-1]['ts'])-datetime.datetime.fromisoformat(ds[0]['ts'])).total_seconds())
  out('低信心',sum(d['decider']=='jev' and (d.get('confidence') or 1)<.35 for d in ds),'最优标记',collections.Counter(d.get('rollout_best_chosen') for d in ds if d['decider']=='jev'),'fallback',sum(bool(d.get('fallback')) for d in ds))
  groups=collections.OrderedDict()
  for d in ds:
   if d.get('turn') and d['screen']=='COMBAT':groups.setdefault((d['floor'],d.get('sl_attempt'),d['turn']),[]).append(d)
  out('代码自主',sum(d['decider']=='code' and d['screen']=='COMBAT' and d['label']!='combat/plan-continue' for d in ds),'回合',len({k for k,v in groups.items() if any(d['decider']=='code' and d['label']!='combat/plan-continue' for d in v)}),'全部战斗回合',len(groups))
  for (f,a,t),v in groups.items():
   s0=state(v[0]);s1=state(v[-1]);c0=compact(s0);c1=compact(s1)
   out('回合',f,a,t,'开始',json.dumps(c0,ensure_ascii=False),'末动作前',json.dumps(c1,ensure_ascii=False),'末',v[-1]['chosen'],v[-1]['result'])
  out('策略')
  for d in ds:
   if d['decider']=='codex':
    out(d['ts'],'F'+str(d['floor']),d['label'],json.dumps(d.get('chosen'),ensure_ascii=False),json.dumps(d.get('journal'),ensure_ascii=False))
  out('特殊干预')
  for d in ds:
   if any(word in d.get('rationale','').lower() for word in ['guard','instead','explor','replay','as good','dominates','sl:','reload']):out(d['ts'],d['floor'],d['turn'],d.get('sl_attempt'),d['label'],d.get('rationale',''))
  out('药水动作')
  for d in ds:
   if 'potion' in str(d.get('chosen')) or d['label']=='reward/claim':out(d['ts'],d['floor'],d['turn'],d.get('sl_attempt'),json.dumps(d.get('chosen'),ensure_ascii=False),d.get('rationale',''))
  out('末状态',json.dumps(compact(ss[-1]['state']),ensure_ascii=False))
  out('末run',json.dumps(ss[-1]['state'].get('run'),ensure_ascii=False))
 print(rid,'已输出',P/(rid+'-summary.txt'))
