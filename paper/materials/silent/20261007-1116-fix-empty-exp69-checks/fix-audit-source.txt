    items.append({'item': item, 'commit': full, 'historical_paths': changed, 'test': test})

for path in files:

for row in items:
    row['base_ancestor'] = git('merge-base', '--is-ancestor', row['commit'], base).returncode == 0
    row['live_ancestor'] = git('merge-base', '--is-ancestor', row['commit'], live).returncode == 0
    row['current_paths'] = sorted({candidate for old in row['historical_paths']
                                   for candidate in by_name.get(Path(old).name, [])})
    assert row['base_ancestor'] and row['live_ancestor'], row
