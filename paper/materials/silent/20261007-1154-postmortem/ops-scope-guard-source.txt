import base64
import collections
import hashlib
import importlib.util
import json
import pathlib
import re
import subprocess

spec = importlib.util.spec_from_file_location('ops_common', '/tmp/sts2-0907-deploy-common.py')
c = importlib.util.module_from_spec(spec)
spec.loader.exec_module(c)
root = c.ROOT
result_path = pathlib.Path('/tmp/sts2-1154-data-result.json')
assert not result_path.exists() and not c.git('diff', '--cached', '--name-only').strip()
receipt = json.loads(pathlib.Path('/tmp/sts2-1154-archive-result.json').read_text())
snap = json.loads(pathlib.Path('/tmp/sts2-1154-paper-result.json').read_text())
assert snap['rc'] == 0
logfile = pathlib.Path('/tmp/sts2-1154-paper-dataset.log')
log = logfile.read_text()
assert 'key scan: CLEAN' in log

def generated(path):
    raw = base64.b64decode(snap['content_base64'][path]) if path in snap['content_base64'] else (root / path).read_bytes()
    assert hashlib.sha256(raw).hexdigest() == snap['after'][path]
    return raw

summary = json.loads(generated('paper/data/summary.json'))
verify = json.loads(generated('paper/data/verification.json'))
flags = {k: v for k, v in verify['checks'].items() if k != 'runs.jsonl decision-count mismatches'}
assert len(flags) == 5 and all(v is True for v in flags.values())
assert verify['checks']['runs.jsonl decision-count mismatches'] == []
cut = re.search(r'cut at (\S+): decisions.jsonl', log).group(1)
assert summary['generated_at'] == cut
changed = []
for path in snap['changed']:
    assert path in snap['content_base64'] and snap['before'].get(path) != snap['after'][path]
    assert path.startswith('paper/data/') or re.fullmatch(r'paper/materials/[^/]+/cost\.md', path)
    raw = generated(path)
    blob = subprocess.run(['git', 'hash-object', '-w', '--path=' + path, '--stdin'], cwd=root,
                          input=raw, stdout=subprocess.PIPE, stderr=subprocess.PIPE, check=True).stdout.decode().strip()
    if blob != c.git('rev-parse', '--verify', ':' + path, check=False).strip():
        c.git('update-index', '--add', '--cacheinfo', '100644,' + blob + ',' + path)
        changed.append(path)

archive = root / receipt['archive']
proof = json.loads((archive / 'manifest.json').read_text())
for entry in proof['raw_manifest']:
    raw = (root / entry['archive']).read_bytes()
    assert (len(raw), hashlib.sha256(raw).hexdigest()) == (entry['bytes'], entry['sha256'])
ledger_path = 'paper/materials/learning/ledger.jsonl'
ledger_base = subprocess.check_output(['git', 'show', 'HEAD:' + ledger_path], cwd=root)
ledger_physical = (root / ledger_path).read_bytes()
foreign = collections.Counter(ledger_physical.splitlines()) - collections.Counter(ledger_base.splitlines())
assert sum(foreign.values()) == receipt['foreign_rows_preserved'] == 3
assert not (collections.Counter(ledger_base.splitlines()) - collections.Counter(ledger_physical.splitlines()))
check = c.run(['nice', '-n', '19', 'python3', 'learner/ledger.py', 'check']).stdout.strip()
assert '223 item(s), 0 problem(s)' in check

now = c.stamp()
ops_manifest = []
rawpaths = []
def capture(origin, name, data=None):
    raw = origin.read_bytes() if data is None else data
    target = archive / name
    assert not target.exists()
    target.write_bytes(raw)
    path = str(target.relative_to(root))
    rawpaths.append(path)
    ops_manifest.append({'source': str(origin), 'archive': path, 'bytes': len(raw),
                         'sha256': hashlib.sha256(raw).hexdigest()})

capture(logfile, 'ops-paper-dataset.txt')
capture(pathlib.Path('/tmp/sts2-1154-archive-result.json'), 'ops-archive-result.json')
capture(pathlib.Path('/tmp/sts2-1154-paper-result.json'), 'ops-paper-capture.json',
        (json.dumps({k: snap[k] for k in ['rc', 'before', 'after', 'changed', 'dirty_before']},
                    ensure_ascii=False, indent=2) + '\n').encode())
manifest_path = archive / 'ops-manifest.json'
assert not manifest_path.exists()
manifest_path.write_text(json.dumps({'recorded': now, 'paper_cut': cut, 'files': ops_manifest},
                                    ensure_ascii=False, indent=2) + '\n')
rawpaths.append(str(manifest_path.relative_to(root)))
c.git('add', '--', *rawpaths)
line = (f'- {now} 运维codex完成11:54复盘批次20261007-114302学习闭环：MCT1GPTL8D35原文含寄生惧魔ID追加勘误及9行旧项update已提交{receipt["commit"]}，'
        '新增纯bug/账本条目无，0079为repeat、其他8项仅support，原shipped/版本/首证/先验/claim保持；打法机制不由运维补写、未记录保持。'
        f'nice19 paper_dataset.py --no-raw exit0，切点{cut}、五校验全过、决策计数差异空、key scan CLEAN，'
        f'行数{json.dumps(verify["row_counts"], ensure_ascii=False, sort_keys=True)}，仅本次{len(changed)}项生成快照及自身原件/记录纳入，{check}；'
        f'另3行并行账本勘误和其他未处理修改保持，无Roy待定事项，详情{receipt["artifact"]}，不停对局/调度、不等待新事件。\n')
c.append_owned(receipt['artifact'], '\n' + line)
c.append_owned(c.LOG, line)
c.append_owned('notes/ops-handoff.md', '\n' + line)
normal = set(changed + [receipt['artifact'], c.LOG, 'notes/ops-handoff.md'])
assert set(c.git('diff', '--cached', '--name-only').splitlines()) == normal | set(rawpaths)
c.git('-c', 'core.whitespace=cr-at-eol', 'diff', '--cached', '--check', '--', *sorted(normal))
c.git('-c', 'core.whitespace=cr-at-eol,-blank-at-eof,-blank-at-eol', 'diff', '--cached', '--check', '--', *rawpaths)
c.run(['nice', '-n', '19', '/home/dw/.local/bin/gitleaks', 'stdin', '--no-banner', '--redact', '--log-level', 'error'],
      data=c.git('diff', '--cached', '--no-ext-diff', '--no-textconv'))
assert (root / ledger_path).read_bytes() == ledger_physical
print(c.git('commit', '-m', 'ops: refresh paper dataset for Silent A10 postmortem\n\n' + c.FOOTER), flush=True)
commit = c.git('rev-parse', 'HEAD').strip()
for path in changed:
    raw = subprocess.check_output(['git', 'show', commit + ':' + path], cwd=root)
    assert hashlib.sha256(raw).hexdigest() == snap['after'][path]
assert not c.git('diff', '--cached', '--name-only').strip()
result = {'commit': commit, 'archive_commit': receipt['commit'], 'cut': cut,
          'generated_paths': changed, 'row_counts': verify['row_counts'], 'ledger_check': check,
          'ledger_rows': 9, 'foreign_rows_preserved': 3, 'new_bugs': 0, 'gitleaks': 0}
result_path.write_text(json.dumps(result, ensure_ascii=False, indent=2) + '\n')
print(json.dumps(result, ensure_ascii=False), flush=True)
