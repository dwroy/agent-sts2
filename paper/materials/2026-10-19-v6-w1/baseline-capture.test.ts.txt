import { writeFileSync } from "node:fs";
import { createHash } from "node:crypto";
import { expect, it, vi } from "vitest";
import { ask, board, choose, decide, env, planned, setupOneshotTests, type Raw } from "./oneshot-support.js";
import { setKnowledgeCharacter } from "../src/knowledge/files.js";
import { runConfigRow } from "../src/eye/run-config.js";
import { loadConfig } from "../src/core/config.js";
vi.mock("../src/knowledge/files.js", async original => ({ ...await original<typeof import("../src/knowledge/files.js")>(),
  knowledgeFile: (_dir: string, name: string) => `/__v6_off_fixed__/${name}` }));
setupOneshotTests();
it("captures fixed baseline bytes on W0, before W1 source changes", () => {
  vi.useFakeTimers(); vi.setSystemTime(new Date("2026-10-19T00:00:00.000Z"));
  const hashes: Record<string, string> = {};
  for (const character of ["ironclad", "silent", "regent"]) {
    setKnowledgeCharacter(character);
    vi.stubEnv("CHARACTER", character);
    for (const [file, key] of [["7b0d-f8-rest", "rest"], ["xljq-f5-reward", "reward"], ["u6ru-f22-shop", "open"]]) {
      const raw = board(file!, key!); Object.assign(raw["run"] as Raw, { character_id: character.toUpperCase(), character_name: character });
      const d = ask(decide(env(raw)));
      const resolved = file!.includes("shop") ? planned(d, []) : choose(d, file!.includes("reward") ? "skip" : "o0");
      hashes[`${character}:${file}`] = createHash("sha256").update(JSON.stringify({ decision: d, resolved })).digest("hex");
    }
    const code = { commit: "0".repeat(40), code: "fixed", dirty: false, dirty_files: [], branch: "fixture", worktree: "fixture" };
    const r = runConfigRow({ env: {}, code, config: loadConfig({ DECISION_LOG: "/__v6_off_fixed__/decisions.jsonl" }), brain: null, jevEnabled: false, mode: "play", knowledgeDir: "/__v6_fixed_missing__", processStarted: "2026-10-19T00:00:00.000Z" },
      { runId: "FIXTURE_RUN", ascension: 0, character: character.toUpperCase(), floor: 1 }, new Date("2026-10-19T00:00:00.000Z"));
    r.process.pid = 0;
    writeFileSync(new URL(`../../.artifacts/run-config-${character}.json`, import.meta.url), JSON.stringify(r, null, 2));
    hashes[`${character}:run-config`] = createHash("sha256").update(JSON.stringify(r)).digest("hex");
  }
  setKnowledgeCharacter(null); vi.unstubAllEnvs(); vi.useRealTimers();
  writeFileSync(new URL("../../.artifacts/v6-off-baseline.json", import.meta.url), JSON.stringify(hashes, null, 2));
  expect(Object.keys(hashes)).toHaveLength(12);
});
