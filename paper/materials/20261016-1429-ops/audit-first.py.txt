sha = lambda raw: hashlib.sha256(raw).hexdigest()


def write(path, value):
    path.write_text(json.dumps(value,ensure_ascii=False,indent=2)+'\n')


def ref(path):
    raw=path.read_bytes()
    return dict(path=str(path.relative_to(r)),bytes=len(raw),sha256=sha(raw))


def report_for(bid):
    output=(r/'ops/codex-ops/learner'/(bid+'.out')).read_text()
    report=json.loads(re.findall(r'```json\s*(\{[\s\S]*?\})\s*```',output)[-1])
    return report,output

                    row=json.loads(raw)

                    assert extract.run_identity(row)==run
                    counts[name]+=1
                    evidence.append(dict(stream=name,offset=item['_offset'],bytes=len(raw),sha256=sha(raw)))

                with (r/'logs'/(name+'.jsonl')).open('rb') as original:
                    for line,raw in enumerate(original,1):
