import difflib
import hashlib
import json
from collections import Counter, defaultdict
from pathlib import Path

S = Path(__file__).resolve().parent
proofs = []
indices = defaultdict(list)
for raw in (S / 'states.jsonl.index.jsonl').open():
    x = json.loads(raw); indices[x['run']].append(x)
states = {}
for run, rows in indices.items():
    local = (S / (run + '-states.jsonl')).read_bytes().splitlines(keepends=True)
    assert len(local) == len(rows)
    attempts = {}; previous = {}
    for raw, ix in zip(local, rows):
        assert hashlib.sha256(raw).hexdigest() == ix['sha256'] and len(raw) == ix['bytes']
        x = json.loads(raw);st=x['state'];f=st['run']['floor'];t=st.get('turn')
        if st.get('combat') and isinstance(t,int) and t>0:
            if f not in attempts: attempts[f]=1
            elif t==1 and previous.get(f,0)>1: attempts[f]+=1
            previous[f]=t
        x['_attempt']=attempts.get(f,0)
        states[ix['line']] = x

def check(name, actual, expected, lines):
    assert actual == expected, (name, actual, expected)
    proofs.append({'claim': name, 'actual': actual, 'expected': expected, 'state_lines': lines})

def hp(line):
    return states[line]['state']['run']['current_hp']

def enemy_hp(line):
    return sum(e['current_hp'] for e in states[line]['state']['combat']['enemies'])

def turn(r, floor, attempt, t):
    st = next(x['state'] for x in states.values() if x['state']['run_id']==r and x['state']['run']['floor']==floor
              and x['_attempt']==attempt and x['state'].get('turn')==t and x['state']['screen']=='COMBAT'
              and (x['state'].get('combat') or {}).get('hand'))
    return {'hp':st['run']['current_hp'], 'enemies':[{'hp':e['current_hp']} for e in st['combat']['enemies']]}
def delta(r, floor, attempt, t):
    a=turn(r,floor,attempt,t);b=turn(r,floor,attempt,t+1)
    return [a['hp']-b['hp'],sum(e['hp'] for e in a['enemies'])-sum(e['hp'] for e in b['enemies'])]

classification=[]
for row in json.loads((S/'source-manifest.json').read_text()):
    stem=Path(row['path']).stem;a=(S/(stem+'-base.txt')).read_text().splitlines();b=(S/(stem+'-live.txt')).read_text().splitlines()
    changed=[line for line in difflib.ndiff(a,b) if line[:2] in ['- ','+ ']]
    minus=[x[2:] for x in changed if x.startswith('- ')];plus=[x[2:] for x in changed if x.startswith('+ ')]
    def author_comment(x,y):
        return x.replace('Roy','Dai')==y and (x.lstrip().startswith(('*','//'))
               or '//' in x and x.index('//')<x.index('Roy'))
    comment_only=not changed or len(minus)==len(plus) and all(author_comment(x,y) for x,y in zip(minus,plus))
    assert comment_only,row['path']
    classification.append({'path':row['path'],'equal':a==b,'only_comment_author_difference':bool(changed),'changed_lines':changed})

(S/'source-difference-classification.json').write_text(json.dumps(classification,ensure_ascii=False,indent=2)+'\n')
(S/'evidence-verification.json').write_text(json.dumps({'scope':'原日志证据算术与来源核验，不是生产源码测试','claims':proofs,'source_rows_verified':len(states),'cunning_actions_verified':len(cunning),'source_comparisons':len(classification)},ensure_ascii=False,indent=2)+'\n')
print('证据核验完成',len(proofs),'项；来源状态',len(states),'帧；狡诈实饮',len(cunning),'次。生产测试未运行。')
