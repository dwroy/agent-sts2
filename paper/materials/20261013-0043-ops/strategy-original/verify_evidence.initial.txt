import hashlib
import json
from collections import Counter
from pathlib import Path

ROOT = Path('/home/dw/Projects/agent-sts2')
OUT = Path(__file__).resolve().parent
PRIOR = ROOT / '.worktrees/codex-strategy-silent-20261012-224301-s2/learner/runs/20261012-224313-strategy-proposal'

def save(name, value):
    (OUT / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')

def digest(raw):
    return hashlib.sha256(raw).hexdigest()

# Prior offsets are locators only. Verify all bytes and identities against the original logs.
indexes = json.loads((PRIOR / 'raw-evidence-union.json').read_text())
identities = json.loads((OUT / 'runs-identity.json').read_text())
extra = {i['run_id'] for i in indexes} - set(identities)
assert extra == {'Y6GM2CHWJBEY'}, extra
with (ROOT / 'logs/runs.jsonl').open('rb') as handle:
    for n, raw in enumerate(handle, 1):
        row = json.loads(raw)
        if row.get('run_id') in extra:
            assert row['character'].lower() == 'silent'
            assert row['run_id'] not in identities
            identities[row['run_id']] = [{'line': n, 'row': row, 'sha256': digest(raw)}]
assert extra <= set(identities)
save('runs-identity.json', identities)
states, decisions, sl = {}, {}, []
manifest = []
for filename in ['states.jsonl', 'decisions.jsonl', 'sl-attempts.jsonl']:
    with (ROOT / 'logs' / filename).open('rb') as source, (OUT / ('raw-' + filename)).open('wb') as target:
        for index in (i for i in indexes if i['file'] == filename):
            source.seek(index['offset'])
            raw = source.read(index['bytes'])
            assert digest(raw) == index['sha256'], index
            row = json.loads(raw)
            actual = row['state']['run_id'] if filename == 'states.jsonl' else row['run_id']
            assert actual == index['run_id'] and actual in identities
            if filename == 'states.jsonl':
                assert row['state']['run']['character_id'].lower() == 'silent'
                states[index['line']] = row
            elif filename == 'decisions.jsonl':
                decisions[index['line']] = row
            else:
                sl.append(row)
            manifest.append({**index, 'saved_offset': target.tell()})
            target.write(raw)
save('raw-evidence-index.json', manifest)
checks = []
used = set()

def state(n):
    used.add(n)
    return states[n]['state']

def hp(n):
    return state(n)['run']['current_hp']

def enemy_hp(n):
    return sum(e['current_hp'] for e in state(n)['combat']['enemies'] if e['is_alive'])

def pair(a, b):
    return [hp(a) - hp(b), enemy_hp(a) - enemy_hp(b)]

def check(name, actual, expected):
    assert actual == expected, (name, actual, expected)
    checks.append({'name': name, 'actual': actual, 'expected': expected})
