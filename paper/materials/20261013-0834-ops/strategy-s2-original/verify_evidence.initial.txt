checks = []


def check(name, observed, expected, lines):
    assert observed == expected, (name, observed, expected)
    checks.append({'name': name, 'observed': observed, 'expected': expected, 'state_lines': lines})


def pp(n, p):
    return f[n]['player_powers'].get(p, 0)


def enemy(n, ident):
    return next(e for e in f[n]['enemies'] if e['id'] == ident)


def ep(n, ident, p):
    return enemy(n, ident)['powers'].get(p, 0)

manifest = json.loads((OUT / 'evidence-manifest.json').read_text())
changed = []
for path, row in manifest['source'].items():
    if row['equal']:
        continue
    base = subprocess.check_output(['git', 'show', manifest['base'] + ':' + path]).decode().splitlines()
    live = subprocess.check_output(['git', 'show', manifest['live_snapshot'] + ':' + path]).decode().splitlines()
    assert len(base) == len(live)
    for old, new in zip(base, live):
        if old != new:
            assert old.lstrip().startswith(('//', '*')) and new == old.replace('Roy', 'Dai'), path
            changed.append({'path': path, 'before': old, 'after': new})
(OUT / 'source-comment-differences.json').write_text(json.dumps(changed, ensure_ascii=False, indent=2) + '\n')
(OUT / 'evidence-checks.json').write_text(json.dumps({'scope': '冻结日志取证算术，未运行生产源码测试',

    'production_tests': {'tsc': None, 'vitest': None, 'cases': None}}, ensure_ascii=False, indent=2) + '\n')
