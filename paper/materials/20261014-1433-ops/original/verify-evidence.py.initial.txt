import hashlib

import pathlib

s = pathlib.Path(__file__).resolve().parent
dispatch = json.loads((s / 'dispatch-and-proposals.json').read_text())
runs = dispatch['dispatch']['runs']
checks = []
def check(label, actual, expected, sources):
    checks.append({'label': label, 'actual': actual, 'expected': expected,
                   'sources': sources, 'passed': actual == expected})

facts = {run: {r['line']: r for r in json.loads((s / (run + '-state-facts.json')).read_text())} for run in runs}
def f(run, line):
    return facts[run][line]
def enemy(row, id):
    return next(e for e in row['enemies'] if e['id'] == id)

# Re-read each selected raw record at its original byte location, including the corrected plan schema.
verified = 0
for kind in ['runs','run-config','states','decisions','sl-attempts','run-plans']:
    index=json.loads((s/(kind+'-index.json')).read_text())
    with (pathlib.Path('/home/dw/Projects/agent-sts2/logs')/(kind+'.jsonl')).open('rb') as source:
        for meta in index:
            source.seek(meta['offset']); raw=source.read(meta['bytes'])
            assert hashlib.sha256(raw).hexdigest()==meta['sha256'], (kind,meta['line'])
            verified += 1
